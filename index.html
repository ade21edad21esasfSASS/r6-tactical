(function() { const canvas = document.getElementById('waveCanvas'); if (!canvas) return; const ctx = canvas.getContext && canvas.getContext('2d'); if (!ctx) return; let width = 0; let height = 0; let time = 0; const waveCount = 5; let waves = []; let rafId = null; let dpr = Math.max(1, window.devicePixelRatio || 1); function resize() { dpr = Math.max(1, window.devicePixelRatio || 1); const cssWidth = window.innerWidth; const cssHeight = window.innerHeight; canvas.style.width = cssWidth + 'px'; canvas.style.height = cssHeight + 'px'; canvas.width = Math.round(cssWidth * dpr); canvas.height = Math.round(cssHeight * dpr); ctx.setTransform(dpr, 0, 0, dpr, 0, 0); // scale drawing ops to CSS pixels width = cssWidth; height = cssHeight; } function initWaves() { waves = []; for (let i = 0; i < waveCount; i++) { waves.push({ amplitude: 15 + Math.random() * 35, frequency: 0.008 + Math.random() * 0.012, speed: 0.4 + Math.random() * 0.6, phase: Math.random() * Math.PI * 2, color: `hsla(${140 + Math.random() * 20}, ${30 + Math.random() * 30}%, ${15 + Math.random() * 20}%, ${0.15 + Math.random() * 0.25})`, thickness: 1.5 + Math.random() * 2 }); } } function drawWaves() { ctx.clearRect(0, 0, width, height); // dark background const gradient = ctx.createLinearGradient(0, 0, 0, height); gradient.addColorStop(0, '#0a1a0f'); gradient.addColorStop(0.4, '#0d1f12'); gradient.addColorStop(0.7, '#0a1a0f'); gradient.addColorStop(1, '#050d08'); ctx.fillStyle = gradient; ctx.fillRect(0, 0, width, height); for (let w = 0; w < waves.length; w++) { const wave = waves[w]; ctx.beginPath(); for (let x = 0; x <= width; x += 1) { const y = height * 0.65 + Math.sin(x * wave.frequency + time * wave.speed + wave.phase) * wave.amplitude + Math.sin(x * wave.frequency * 2.3 + time * wave.speed * 0.7 + wave.phase * 1.4) * (wave.amplitude * 0.3) + Math.sin(x * wave.frequency * 0.5 + time * wave.speed * 0.3 + wave.phase * 0.8) * (wave.amplitude * 0.5); if (x === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y); } const grad = ctx.createLinearGradient(0, height * 0.5, 0, height); const alpha = 0.15 + (w / waves.length) * 0.25; grad.addColorStop(0, `rgba(20, 50, 30, ${alpha * 0.3})`); grad.addColorStop(0.5, `rgba(15, 40, 25, ${alpha * 0.6})`); grad.addColorStop(1, `rgba(5, 15, 10, ${alpha * 0.9})`); ctx.lineTo(width, height); ctx.lineTo(0, height); ctx.closePath(); ctx.fillStyle = grad; ctx.fill(); ctx.strokeStyle = wave.color; ctx.lineWidth = wave.thickness; ctx.shadowColor = 'rgba(29, 185, 84, 0.05)'; ctx.shadowBlur = 10; ctx.stroke(); ctx.shadowBlur = 0; if (w % 2 === 0) { for (let x = 0; x < width; x += 40) { const y = height * 0.65 + Math.sin(x * wave.frequency + time * wave.speed + wave.phase) * wave.amplitude + Math.sin(x * wave.frequency * 2.3 + time * wave.speed * 0.7 + wave.phase * 1.4) * (wave.amplitude * 0.3); ctx.fillStyle = `rgba(100, 220, 140, ${0.12})`; ctx.beginPath(); ctx.arc(x, y - 2, 1.5, 0, Math.PI * 2); ctx.fill(); } } } // fog const fogGrad = ctx.createLinearGradient(0, height * 0.8, 0, height); fogGrad.addColorStop(0, 'rgba(10, 20, 15, 0)'); fogGrad.addColorStop(1, 'rgba(5, 15, 10, 0.5)'); ctx.fillStyle = fogGrad; ctx.fillRect(0, height * 0.8, width, height * 0.2); time += 0.008; rafId = requestAnimationFrame(drawWaves); } // visibility handling function handleVisibility() { if (document.hidden) { if (rafId) { cancelAnimationFrame(rafId); rafId = null; } } else { if (!rafId) drawWaves(); } } // debounce resize let resizeTimer = null; function onResize() { clearTimeout(resizeTimer); resizeTimer = setTimeout(() => { resize(); initWaves(); }, 150); } window.addEventListener('resize', onResize, { passive: true }); document.addEventListener('visibilitychange', handleVisibility); // init resize(); initWaves(); drawWaves(); })();
